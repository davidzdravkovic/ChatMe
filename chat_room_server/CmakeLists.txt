cmake_minimum_required(VERSION 3.20)
project(chat_server)

option(CHAT_ENABLE_JWT "Link OpenSSL + jwt-cpp for real HS256 JWT" ON)
option(CHAT_ENABLE_PASSWORD_HASH "Hash passwords with libsodium (Argon2id)" ON)

# Compiler needs to use C++17 not silently fail on older version, standard version no extensions
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---- Find dependencies ----
find_package(PostgreSQL REQUIRED)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
set(Boost_NO_BOOST_CMAKE ON)
find_package(Boost REQUIRED)

if(CHAT_ENABLE_PASSWORD_HASH)
  # Prefer pkg-config (libsodium.pc). No built-in FindSodium / Config package required.
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SODIUM QUIET libsodium)
  endif()
  if(SODIUM_FOUND)
    set(SODIUM_INCLUDE_DIR ${SODIUM_INCLUDE_DIRS})
    set(SODIUM_LIBRARY ${SODIUM_LIBRARIES})
  else()
    find_path(SODIUM_INCLUDE_DIR sodium.h)
    find_library(SODIUM_LIBRARY NAMES sodium libsodium)
  endif()
  if(NOT SODIUM_INCLUDE_DIR OR NOT SODIUM_LIBRARY)
    message(FATAL_ERROR
      "libsodium not found (needed for CHAT_ENABLE_PASSWORD_HASH). "
      "Linux: sudo apt install -y libsodium-dev pkg-config. "
      "MSYS2 MinGW: pacman -S mingw-w64-x86_64-libsodium mingw-w64-x86_64-pkgconf. "
      "Or disable: -DCHAT_ENABLE_PASSWORD_HASH=OFF")
  endif()
endif()

find_package(OpenSSL REQUIRED)

if(CHAT_ENABLE_JWT)
 # Linux has no package for jwt-cpp so locally fetch to the working tree if there is not found in the working directory
  if(UNIX AND NOT APPLE)
    include(FetchContent)
    message(STATUS "Using FetchContent for jwt-cpp (Linux/BSD)")
    set(JWTCPP_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
      jwt-cpp
      GIT_REPOSITORY https://github.com/Thalhammer/jwt-cpp.git
      GIT_TAG v0.7.1
    )
    FetchContent_MakeAvailable(jwt-cpp)
  else()
    find_package(jwt-cpp CONFIG REQUIRED)
  endif()
endif()

# Downloads nLohmann a json library one file only if there is not found in the working directory
set(JSON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann_json")
set(JSON_HPP  "${JSON_DIR}/include/nlohmann/json.hpp")
if(NOT EXISTS "${JSON_HPP}")
  message(STATUS "Downloading nlohmann/json into third_party/nlohmann_json ...")
  file(MAKE_DIRECTORY "${JSON_DIR}/include/nlohmann")
  file(DOWNLOAD
    "https://github.com/nlohmann/json/releases/download/v3.11.3/json.hpp"
    "${JSON_HPP}"
    TLS_VERIFY ON
    STATUS _status
  )
  list(GET _status 0 _code)
  if(NOT _code EQUAL 0)
    message(FATAL_ERROR "Failed to download nlohmann/json.hpp")
  endif()
endif()

# ---- Sources ----
# CONFIGURE_DEPENDS: re-scan when .cpp files are added/removed (plain GLOB is stale until reconfigure).
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/*.cpp")
# If GLOB was evaluated before a new .cpp existed, CMakeCache keeps a stale list — append known files once.
foreach(
    _extra
    "${PROJECT_SOURCE_DIR}/src/Network/RequestPipeline.cpp"
)
  if(EXISTS "${_extra}")
    list(FIND SOURCES "${_extra}" _idx)
    if(_idx EQUAL -1)
      list(APPEND SOURCES "${_extra}")
    endif()
  endif()
endforeach()

add_executable(${PROJECT_NAME} ${SOURCES})

# ---- Shared request/response DTOs (copy or symlink `common` next to this CMakeLists.txt) ----
set(CHAT_COMMON_DIR "${PROJECT_SOURCE_DIR}/common")

# ---- Includes ----
target_include_directories(${PROJECT_NAME} PRIVATE
    ${PROJECT_SOURCE_DIR}/include
    "${JSON_DIR}/include"
    "${CHAT_COMMON_DIR}"
    ${Boost_INCLUDE_DIRS}
    $<$<BOOL:${CHAT_ENABLE_PASSWORD_HASH}>:${SODIUM_INCLUDE_DIR}>
)

# ---- Link libraries ----
  target_link_libraries(${PROJECT_NAME} PRIVATE
      pqxx
      PostgreSQL::PostgreSQL
  )


if(UNIX AND NOT APPLE)
  target_link_libraries(${PROJECT_NAME} PRIVATE pthread)
endif()

if(CHAT_ENABLE_PASSWORD_HASH)
  target_link_libraries(${PROJECT_NAME} PRIVATE ${SODIUM_LIBRARY})
  target_compile_definitions(${PROJECT_NAME} PRIVATE CHAT_ENABLE_PASSWORD_HASH=1)
endif()

if(CHAT_ENABLE_JWT)
  target_link_libraries(${PROJECT_NAME} PRIVATE
    OpenSSL::SSL
    OpenSSL::Crypto
    jwt-cpp::jwt-cpp
  )
  target_compile_definitions(${PROJECT_NAME} PRIVATE CHAT_ENABLE_JWT=1)
else()
  target_link_libraries(${PROJECT_NAME} PRIVATE OpenSSL::Crypto)
endif()

if (WIN32)
    target_link_libraries(${PROJECT_NAME} PRIVATE
        ws2_32
        Mswsock
    )
endif()

set_target_properties(${PROJECT_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
)
