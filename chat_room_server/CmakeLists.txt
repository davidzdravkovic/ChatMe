cmake_minimum_required(VERSION 3.20)
project(chat_server)

# JWT (OpenSSL + jwt-cpp): OFF by default so existing MSYS2-only configures keep working.
# Linux: sudo apt install -y libssl-dev libjwt-cpp-dev  (or only libssl-dev — CMake can FetchContent jwt-cpp).
# Windows: -DCHAT_ENABLE_JWT=ON with vcpkg / CMake prefix where jwt-cpp is installed.
option(CHAT_ENABLE_JWT "Link OpenSSL + jwt-cpp for real HS256 JWT" OFF)
option(CHAT_ENABLE_PASSWORD_HASH "Hash passwords with libsodium (Argon2id)" ON)
# OFF = old CMake style: find_package(Boost) headers only, link pqxx only (no Boost::system/thread).
# Asio/Beast normally still need libboost_system at link time — expect link errors unless you use b2/MSYS2 libs.
option(CHAT_LINK_BOOST_LIBRARIES "Link Boost system+thread (ON recommended for this codebase)" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---- Platform-specific paths ----
# Linux/VPS: no hardcoded roots — apt packages satisfy find_package(Boost), PostgreSQL, etc.
# Windows (MSYS2 MinGW): install deps in one prefix, e.g.
#   pacman -S mingw-w64-x86_64-boost mingw-w64-x86_64-postgresql mingw-w64-x86_64-libpqxx
# Optional: cmake -DCHAT_BOOST_ROOT=C:/path/to/boost/prefix if you use a custom Boost build.
# Windows: BOOST_ROOT env + optional -DCHAT_BOOST_ROOT. Supports:
#   - install layout: <prefix>/include/boost/version.hpp
#   - source tarball layout: <prefix>/boost/version.hpp (same as your old working CMake)
set(CHAT_BOOST_ROOT "" CACHE PATH "Windows only: Boost prefix (include/ or boost/ at root). Empty = use env / legacy path / MSYS2.")

if(WIN32)
  set(_CHAT_LEGACY_BOOST "C:/Users/User/librariesboost/boost_1_89_0")
  set(_CHAT_MSYS2_MINGW64 "C:/msys64/mingw64")
  if(EXISTS "${_CHAT_MSYS2_MINGW64}")
    list(APPEND CMAKE_PREFIX_PATH "${_CHAT_MSYS2_MINGW64}")
  endif()

  # ---- Headers-only mode (matches your old CMake; no pacman / no b2 required for *configure*) ----
  if(NOT CHAT_LINK_BOOST_LIBRARIES)
    if(CHAT_BOOST_ROOT)
      set(Boost_ROOT "${CHAT_BOOST_ROOT}" CACHE PATH "Boost (CHAT_BOOST_ROOT)" FORCE)
      list(APPEND CMAKE_PREFIX_PATH "${CHAT_BOOST_ROOT}")
    elseif(DEFINED ENV{BOOST_ROOT})
      cmake_path(SET _CHAT_ENV_BOOST "$ENV{BOOST_ROOT}")
      cmake_path(NORMAL_PATH _CHAT_ENV_BOOST)
      if(EXISTS "${_CHAT_ENV_BOOST}/include/boost/version.hpp" OR EXISTS "${_CHAT_ENV_BOOST}/boost/version.hpp")
        set(Boost_ROOT "${_CHAT_ENV_BOOST}" CACHE PATH "Boost (BOOST_ROOT env)" FORCE)
        list(APPEND CMAKE_PREFIX_PATH "${_CHAT_ENV_BOOST}")
      endif()
    elseif(EXISTS "${_CHAT_LEGACY_BOOST}/boost/version.hpp")
      set(Boost_ROOT "${_CHAT_LEGACY_BOOST}" CACHE PATH "Boost (legacy tarball headers)" FORCE)
      list(APPEND CMAKE_PREFIX_PATH "${_CHAT_LEGACY_BOOST}")
    endif()
  elseif(CHAT_BOOST_ROOT)

  
    set(Boost_ROOT "${CHAT_BOOST_ROOT}" CACHE PATH "Boost install prefix" FORCE)
    list(APPEND CMAKE_PREFIX_PATH "${CHAT_BOOST_ROOT}")
  elseif(DEFINED ENV{BOOST_ROOT})
    cmake_path(SET _CHAT_ENV_BOOST "$ENV{BOOST_ROOT}")
    cmake_path(NORMAL_PATH _CHAT_ENV_BOOST)
    if(EXISTS "${_CHAT_ENV_BOOST}/include/boost/version.hpp" OR EXISTS "${_CHAT_ENV_BOOST}/boost/version.hpp")
      set(Boost_ROOT "${_CHAT_ENV_BOOST}" CACHE PATH "Boost (BOOST_ROOT environment variable)" FORCE)
      list(APPEND CMAKE_PREFIX_PATH "${_CHAT_ENV_BOOST}")
    endif()
  else()
    # MSYS2: one prefix for headers + libs (pacman: mingw-w64-x86_64-boost).
    set(_CHAT_MSVC_SYS_LIBS FALSE)
    file(GLOB _CHAT_MSYS_BSYS
      "${_CHAT_MSYS2_MINGW64}/lib/libboost_system*.dll.a"
      "${_CHAT_MSYS2_MINGW64}/lib/libboost_system*.a")
    list(LENGTH _CHAT_MSYS_BSYS _CHAT_MSYS_BSYS_N)
    if(EXISTS "${_CHAT_MSYS2_MINGW64}/include/boost/version.hpp")
      set(Boost_ROOT "${_CHAT_MSYS2_MINGW64}" CACHE PATH "Boost (MSYS2 mingw64)" FORCE)
      set(_CHAT_MSVC_SYS_LIBS TRUE)
    elseif(_CHAT_MSYS_BSYS_N GREATER 0)
      message(FATAL_ERROR
        "Found Boost libraries under ${_CHAT_MSYS2_MINGW64}/lib but no include/boost headers.\n"
        "Install the dev package (headers):  pacman -S mingw-w64-x86_64-boost\n"
        "Or your MSYS2 prefix may differ from C:/msys64/mingw64 — set -DCHAT_BOOST_ROOT= to the correct prefix.")
    endif()
    if(NOT _CHAT_MSVC_SYS_LIBS)
      if(EXISTS "${_CHAT_LEGACY_BOOST}/boost/version.hpp" OR EXISTS "${_CHAT_LEGACY_BOOST}/include/boost/version.hpp")
        file(GLOB _CHAT_LEG_BSYS
          "${_CHAT_LEGACY_BOOST}/stage/lib/libboost_system*.dll.a"
          "${_CHAT_LEGACY_BOOST}/stage/lib/libboost_system*.a"
          "${_CHAT_LEGACY_BOOST}/lib/libboost_system*.dll.a"
          "${_CHAT_LEGACY_BOOST}/lib/libboost_system*.a")
        list(LENGTH _CHAT_LEG_BSYS _CHAT_LEG_BSYS_N)
        if(_CHAT_LEG_BSYS_N GREATER 0)
          set(Boost_ROOT "${_CHAT_LEGACY_BOOST}" CACHE PATH "Boost (legacy tree with stage/lib)" FORCE)
          list(APPEND CMAKE_PREFIX_PATH "${_CHAT_LEGACY_BOOST}")
        endif()
      endif()
    endif()
  endif()
endif()

# ---- Find dependencies FIRST ----
if(WIN32)
  set(_CHAT_LEGACY_CHECK "C:/Users/User/librariesboost/boost_1_89_0")
  set(_CHAT_MSYS2_ROOT "C:/msys64/mingw64")
  set(_CHAT_HAVE_HEADERS FALSE)
  if(CHAT_BOOST_ROOT AND (
      EXISTS "${CHAT_BOOST_ROOT}/include/boost/version.hpp" OR
      EXISTS "${CHAT_BOOST_ROOT}/boost/version.hpp"))
    set(_CHAT_HAVE_HEADERS TRUE)
  elseif(DEFINED ENV{BOOST_ROOT})
    cmake_path(SET _CHAT_EBC "$ENV{BOOST_ROOT}")
    cmake_path(NORMAL_PATH _CHAT_EBC)
    if(EXISTS "${_CHAT_EBC}/include/boost/version.hpp" OR EXISTS "${_CHAT_EBC}/boost/version.hpp")
      set(_CHAT_HAVE_HEADERS TRUE)
    endif()
  elseif(EXISTS "${_CHAT_LEGACY_CHECK}/boost/version.hpp" OR EXISTS "${_CHAT_LEGACY_CHECK}/include/boost/version.hpp")
    set(_CHAT_HAVE_HEADERS TRUE)
  elseif(EXISTS "${_CHAT_MSYS2_ROOT}/include/boost/version.hpp")
    set(_CHAT_HAVE_HEADERS TRUE)
  endif()
  if(NOT _CHAT_HAVE_HEADERS)
    message(FATAL_ERROR
      "Could not find Boost headers on Windows.\n"
      "Install: pacman -S mingw-w64-x86_64-boost  OR set CHAT_BOOST_ROOT / BOOST_ROOT to a Boost prefix.")
  endif()
  # When linking Boost libs: need a prefix that contains built system/thread (MSYS2 or b2 stage/lib).
  if(CHAT_LINK_BOOST_LIBRARIES AND NOT Boost_ROOT)
    file(GLOB _CHAT_ANY_SYS
      "${_CHAT_MSYS2_ROOT}/lib/libboost_system*.dll.a"
      "${_CHAT_MSYS2_ROOT}/lib/libboost_system*.a"
      "${_CHAT_LEGACY_CHECK}/stage/lib/libboost_system*.dll.a"
      "${_CHAT_LEGACY_CHECK}/stage/lib/libboost_system*.a")
    list(LENGTH _CHAT_ANY_SYS _CHAT_ANY_SYS_N)
    if(_CHAT_ANY_SYS_N EQUAL 0)
      message(FATAL_ERROR
        "Boost *libraries* not found (need system + thread for this server).\n"
        "CMake would only see headers from ${_CHAT_LEGACY_CHECK} without built libs.\n"
        "Do ONE of:\n"
        "  1) MSYS2 MinGW64:  pacman -S mingw-w64-x86_64-boost\n"
        "  2) In your Boost tree:  ./bootstrap.sh && ./b2 --with-system --with-thread\n"
        "Or configure with -DCHAT_LINK_BOOST_LIBRARIES=OFF (old CMake style; link step may still fail).\n"
        "Then delete build/ and run cmake again.")
    endif()
  endif()
endif()

find_package(OpenSSL REQUIRED)

if(CHAT_ENABLE_JWT)
  # Linux: never require a system jwt-cpp package (Noble has no libjwt-cpp-dev); fetch header library.
  if(UNIX AND NOT APPLE)
    include(FetchContent)
    message(STATUS "Using FetchContent for jwt-cpp (Linux/BSD)")
    set(JWTCPP_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
      jwt-cpp
      GIT_REPOSITORY https://github.com/Thalhammer/jwt-cpp.git
      GIT_TAG v0.7.1
    )
    FetchContent_MakeAvailable(jwt-cpp)
  else()
    find_package(jwt-cpp CONFIG REQUIRED)
  endif()

# ---- nlohmann/json (header-only, in project tree) ----
set(JSON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann_json")
set(JSON_HPP  "${JSON_DIR}/include/nlohmann/json.hpp")
if(NOT EXISTS "${JSON_HPP}")
  message(STATUS "Downloading nlohmann/json into third_party/nlohmann_json ...")
  file(MAKE_DIRECTORY "${JSON_DIR}/include/nlohmann")
  file(DOWNLOAD
    "https://github.com/nlohmann/json/releases/download/v3.11.3/json.hpp"
    "${JSON_HPP}"
    TLS_VERIFY ON
    STATUS _status
  )
  list(GET _status 0 _code)
  if(NOT _code EQUAL 0)
    message(FATAL_ERROR "Failed to download nlohmann/json.hpp")
  endif()
endif()

# ---- Sources ----
# CONFIGURE_DEPENDS: re-scan when .cpp files are added/removed (plain GLOB is stale until reconfigure).
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/*.cpp")
# If GLOB was evaluated before a new .cpp existed, CMakeCache keeps a stale list — append known files once.
foreach(
    _extra
    "${PROJECT_SOURCE_DIR}/src/Network/RequestPipeline.cpp"
)
  if(EXISTS "${_extra}")
    list(FIND SOURCES "${_extra}" _idx)
    if(_idx EQUAL -1)
      list(APPEND SOURCES "${_extra}")
    endif()
  endif()
endforeach()

add_executable(${PROJECT_NAME} ${SOURCES})

# ---- Shared request/response DTOs (copy or symlink `common` next to this CMakeLists.txt) ----
set(CHAT_COMMON_DIR "${PROJECT_SOURCE_DIR}/common")
